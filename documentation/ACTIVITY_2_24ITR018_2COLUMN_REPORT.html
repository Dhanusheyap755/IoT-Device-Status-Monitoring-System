backend/fsm/deviceStateMachine.js
/**
 * FSM Module - FLAT DFA Engine
 * M = (Q, Σ, δ, q0, F)
 * Q = {Registered, Active, Disconnected}
 * Σ = {ValidMessage, InvalidMessage, CommunicationFailure, Timeout, Reconnect}
 * q0 = Registered, F = {Active}
 */
const DeviceEvent = require('../models/DeviceEvent');
const Alert = require('../models/Alert');

const STATES = { REGISTERED: 'Registered', ACTIVE: 'Active', DISCONNECTED: 'Disconnected' };
const EVENTS = {
  VALID_MESSAGE: 'ValidMessage',
  INVALID_MESSAGE: 'InvalidMessage',
  COMMUNICATION_FAILURE: 'CommunicationFailure',
  TIMEOUT: 'Timeout',
  RECONNECT: 'Reconnect',
};

const TRANSITION_TABLE = {
  [STATES.REGISTERED]: {
    [EVENTS.VALID_MESSAGE]: { nextState: STATES.ACTIVE, action: 'Promote to Active and save reading', createsAlert: false },
    [EVENTS.INVALID_MESSAGE]: { nextState: STATES.REGISTERED, action: 'Maintain Registered, reject payload, log alert', createsAlert: true, alertType: 'INVALID_MESSAGE', severity: 'Warning' },
    [EVENTS.COMMUNICATION_FAILURE]: { nextState: STATES.DISCONNECTED, action: 'Demote to Disconnected due to outage', createsAlert: true, alertType: 'COMMUNICATION_FAILURE', severity: 'Error' },
    [EVENTS.TIMEOUT]: { nextState: STATES.DISCONNECTED, action: 'Demote to Disconnected on timeout', createsAlert: true, alertType: 'TIMEOUT', severity: 'Warning' },
    [EVENTS.RECONNECT]: { nextState: STATES.ACTIVE, action: 'Activate via operator reconnection', createsAlert: true, alertType: 'RECONNECTION', severity: 'Info' },
  },
  [STATES.ACTIVE]: {
    [EVENTS.VALID_MESSAGE]: { nextState: STATES.ACTIVE, action: 'Maintain Active (Self Loop), save telemetry', createsAlert: false },
    [EVENTS.INVALID_MESSAGE]: { nextState: STATES.ACTIVE, action: 'Maintain Active (Self Loop), reject payload, log alert', createsAlert: true, alertType: 'INVALID_MESSAGE', severity: 'Warning' },
    [EVENTS.COMMUNICATION_FAILURE]: { nextState: STATES.DISCONNECTED, action: 'Demote to Disconnected on outage', createsAlert: true, alertType: 'COMMUNICATION_FAILURE', severity: 'Error' },
    [EVENTS.TIMEOUT]: { nextState: STATES.DISCONNECTED, action: 'Demote to Disconnected on keepalive timeout', createsAlert: true, alertType: 'TIMEOUT', severity: 'Warning' },
    [EVENTS.RECONNECT]: { nextState: STATES.ACTIVE, action: 'Reaffirm Active state', createsAlert: true, alertType: 'RECONNECTION', severity: 'Info' },
  },
  [STATES.DISCONNECTED]: {
    [EVENTS.VALID_MESSAGE]: { nextState: STATES.ACTIVE, action: 'Auto-reconnect to Active upon valid message', createsAlert: true, alertType: 'RECONNECTION', severity: 'Info' },
    [EVENTS.INVALID_MESSAGE]: { nextState: STATES.DISCONNECTED, action: 'Maintain Disconnected (Self Loop), reject payload', createsAlert: true, alertType: 'INVALID_MESSAGE', severity: 'Warning' },
    [EVENTS.COMMUNICATION_FAILURE]: { nextState: STATES.DISCONNECTED, action: 'Maintain Disconnected, log error alert', createsAlert: true, alertType: 'COMMUNICATION_FAILURE', severity: 'Error' },
    [EVENTS.TIMEOUT]: { nextState: STATES.DISCONNECTED, action: 'Maintain Disconnected, log timeout alert', createsAlert: true, alertType: 'TIMEOUT', severity: 'Warning' },
    [EVENTS.RECONNECT]: { nextState: STATES.ACTIVE, action: 'Restore to Active via operator reconnect', createsAlert: true, alertType: 'RECONNECTION', severity: 'Info' },
  },
};

function evaluateTransition(currentState, event) {
  if (!TRANSITION_TABLE[currentState]) throw new Error(`Invalid state: "${currentState}"`);
  const transition = TRANSITION_TABLE[currentState][event];
  if (!transition) throw new Error(`Invalid event: "${event}" for state "${currentState}"`);
  return { previousState: currentState, event, nextState: transition.nextState, stateChanged: currentState !== transition.nextState, action: transition.action, createsAlert: transition.createsAlert, alertType: transition.alertType || null, severity: transition.severity || 'Info' };
}

async function processFSMTransition(device, event, options = {}) {
  const currentState = device.state;
  const evaluation = evaluateTransition(currentState, event);
  const transitionTimestamp = new Date();
  const previousState = device.state;
  device.state = evaluation.nextState;
  device.lastSeen = transitionTimestamp;
  if (options.temperature !== undefined && options.temperature !== null) device.temperature = options.temperature;
  if (options.humidity !== undefined && options.humidity !== null) device.humidity = options.humidity;
  if (typeof device.save === 'function') await device.save();

  const eventRecord = new DeviceEvent({
    deviceId: device.deviceId, previousState, event, currentState: evaluation.nextState,
    details: { action: evaluation.action, stateChanged: evaluation.stateChanged, reason: options.reason || null, ...options.metadata },
    timestamp: transitionTimestamp,
  });
  await eventRecord.save();

  let alertRecord = null;
  if (evaluation.createsAlert || options.forceAlert) {
    alertRecord = new Alert({
      deviceId: device.deviceId, alertType: options.alertType || evaluation.alertType || 'SYSTEM_INFO',
      message: options.alertMessage || `Device ${device.deviceId} triggered event '${event}'. State is now [${evaluation.nextState}].`,
      severity: options.severity || evaluation.severity || 'Warning',
      details: { previousState, currentState: evaluation.nextState, event, reason: options.reason || null, ...options.metadata },
      createdAt: transitionTimestamp,
    });
    await alertRecord.save();
  }
  return { success: true, deviceId: device.deviceId, previousState, event, currentState: evaluation.nextState, stateChanged: evaluation.stateChanged, action: evaluation.action, timestamp: transitionTimestamp, alert: alertRecord };
}

module.exports = { STATES, EVENTS, TRANSITION_TABLE, evaluateTransition, processFSMTransition };

const validateSensorMessage = async (req, res, next) => {
  const { deviceId, temperature, humidity, status } = req.body || {};
  const validationErrors = [];

  if (!deviceId || typeof deviceId !== 'string' || deviceId.trim() === '') {
    validationErrors.push('Device ID is mandatory.');
  } else if (!/^IOT\d{3,}$/.test(deviceId.trim().toUpperCase())) {
    validationErrors.push(`Device ID "${deviceId}" is invalid. Must match IOT001.`);
  }

  if (temperature === undefined || temperature === null || temperature === '') {
    validationErrors.push('Temperature reading is mandatory.');
  } else {
    const tempNum = Number(temperature);
    if (isNaN(tempNum) || typeof temperature === 'boolean') validationErrors.push(`Temperature must be a valid number.`);
    else if (tempNum < -40 || tempNum > 125) validationErrors.push(`Temperature (${tempNum}°C) is out of range (-40°C to 125°C).`);
  }

  if (humidity === undefined || humidity === null || humidity === '') {
    validationErrors.push('Humidity reading is mandatory.');
  } else {
    const humNum = Number(humidity);
    if (isNaN(humNum) || typeof humidity === 'boolean') validationErrors.push(`Humidity must be a valid number.`);
    else if (humNum < 0 || humNum > 100) validationErrors.push(`Humidity (${humNum}%) is out of range (0% to 100%).`);
  }

  if (!status || (status.trim().toLowerCase() !== 'online' && status.trim().toLowerCase() !== 'offline')) {
    validationErrors.push('Status field must be "online" or "offline".');
  }

  if (validationErrors.length > 0) {
    const targetId = typeof deviceId === 'string' && deviceId.trim() ? deviceId.trim().toUpperCase() : 'UNKNOWN';
    let dev = null;
    if (targetId !== 'UNKNOWN') { try { dev = await Device.findOne({ deviceId: targetId }); } catch(e){} }
    if (dev) {
      await processFSMTransition(dev, EVENTS.INVALID_MESSAGE, {
        reason: validationErrors.join(' | '),
        alertMessage: `Validation failed for ${targetId}: ${validationErrors.join('; ')}`,
        severity: 'Warning', alertType: 'INVALID_MESSAGE', metadata: { rejectedPayload: req.body, errors: validationErrors }
      });
    } else {
      await Alert.create({ deviceId: targetId, alertType: 'INVALID_MESSAGE', message: `Malformed message: ${validationErrors.join('; ')}`, severity: 'Error', details: { rejectedPayload: req.body } });
    }
    return res.status(400).json({ success: false, error: 'Validation Failed', deviceId: targetId, validationErrors, message: 'Payload rejected. State unchanged.' });
  }

  const cleanId = deviceId.trim().toUpperCase();
  try {
    const dev = await Device.findOne({ deviceId: cleanId });
    if (!dev) return res.status(404).json({ success: false, error: 'Unknown Device ID', message: `Device ${cleanId} is not registered.` });
    req.targetDevice = dev;
    req.sanitizedPayload = { deviceId: cleanId, temperature: Number(temperature), humidity: Number(humidity), status: status.trim().toLowerCase() };
    next();
  } catch(err) {
    return res.status(503).json({ success: false, error: 'Database Unavailable' });
  }
};
module.exports = { validateSensorMessage };

const registerDevice = async (req, res, next) => {
  try {

    await Alert.create({ deviceId: fId, alertType: 'REGISTRATION', message: `Device ${fId} registered. Initial state [Registered].`, severity: 'Info' });

    res.status(200).json({ success: true, data: { deviceId: cleanId, currentState: transition.currentState } });
  } catch(e) { next(e); }
};

const getDeviceHistory = async (req, res, next) => {
  try {
    const id = req.params.id.trim().toUpperCase();
    const history = await DeviceEvent.find({ deviceId: id }).sort({ timestamp: -1 });
    res.status(200).json({ success: true, deviceId: id, data: history });
  } catch(e) { next(e); }
};

const getSystemStats = async (req, res, next) => {
  try {
    const [total, active, disconnected, registered, readings, alerts] = await Promise.all([
      Device.countDocuments(), Device.countDocuments({ state: STATES.ACTIVE }),
      Device.countDocuments({ state: STATES.DISCONNECTED }), Device.countDocuments({ state: STATES.REGISTERED }),
      SensorReading.countDocuments(), Alert.countDocuments()
    ]);
    res.status(200).json({ success: true, data: { summary: { totalDevices: total, activeDevices: active, disconnectedDevices: disconnected, registeredDevices: registered, totalReadings: readings, totalAlerts: alerts } } });
  } catch(e) { next(e); }
};

const rawApiUrl = import.meta.env.VITE_API_BASE_URL || import.meta.env.VITE_API_URL || 'http://localhost:5000/api';
const cleanUrl = rawApiUrl.replace(/\/+$/, '');
export const API_BASE_URL = cleanUrl.endsWith('/api') ? cleanUrl : `${cleanUrl}/api`;

async function request(endpoint, options = {}) {
  const formatted = endpoint.startsWith('/') ? endpoint : `/${endpoint}`;
  const url = `${API_BASE_URL}${formatted}`;
  const config = { headers: { 'Content-Type': 'application/json', ...options.headers }, ...options };
  try {
    const response = await fetch(url, config);
    const data = await response.json().catch(() => ({}));
    if (!response.ok) {
      const err = new Error(data.message || data.error || `HTTP ${response.status}`);
      err.status = response.status; err.data = data;
      throw err;
    }
    return data;
  } catch (err) {
    if (err.name === 'TypeError' && err.message.includes('fetch')) {
      const connErr = new Error(`Unable to reach backend API at ${API_BASE_URL}.`);
      connErr.status = 0; throw connErr;
    }
    throw err;
  }
}
